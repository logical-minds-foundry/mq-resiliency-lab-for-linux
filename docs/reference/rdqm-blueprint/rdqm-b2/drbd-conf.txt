### /etc/drbd.d/rdqmapp.dr.res
resource rdqmapp.dr {
    options {
        quorum majority;
        on-no-quorum io-error;
        auto-promote no;
    }
    volume 0 {
        device minor 7001;
    }
    handlers {
      before-resync-source "/var/mqm/rdqm/before-resync-source";
      before-resync-target "/var/mqm/rdqm/before-resync-target";
      after-resync-target "/var/mqm/rdqm/after-resync-target";
      disconnected "/var/mqm/rdqm/disconnected";
    }
    stacked-on-top-of rdqmapp {
        node-id 1;
    }
    on encrypted.remote {
        node-id 0;
        volume 0 {
            disk _remote;
        }
    }
    connection {
        net {
            protocol A;
            verify-alg sha1;
            allow-remote-read no;
            tls yes;
        }
        path {
            host rdqm-b2 address ipv4 10.99.0.42:7001;
            host encrypted.remote address ipv4 10.99.0.31:7001;
        }
        path {
            host rdqm-b2 address ipv4 10.99.0.42:7001;
            host encrypted.remote address ipv4 10.99.0.32:7001;
        }
        path {
            host rdqm-b2 address ipv4 10.99.0.42:7001;
            host encrypted.remote address ipv4 10.99.0.33:7001;
        }
    }
}
### /etc/drbd.d/rdqmapp.res
resource rdqmapp {
    options {
        quorum 2;
        on-no-quorum io-error;
        auto-promote no;
    }
    net {
        verify-alg sha1;
        rr-conflict retry-connect;
        tls yes;
        #
    }
    disk {
        disable-write-same yes;
    }
    handlers {
      quorum-lost "/var/mqm/rdqm/quorum-lost";
      after-resync-target "/var/mqm/rdqm/after-resync-target";
      disconnected "/var/mqm/rdqm/disconnected";
    }
    on rdqm-b1 {
        node-id 0;
        address ipv4 172.16.2.41:7000;
        volume 0 {
            device minor 100;
            disk /dev/drbdpool/rdqmapp_00;
            meta-disk internal;
        }
    }
   on rdqm-b2 {
       node-id 1;
       address ipv4 172.16.2.42:7000;
       volume 0 {
            device minor 100;
            disk /dev/drbdpool/rdqmapp_00;
            meta-disk internal;
        }
    }
    on rdqm-b3 {
        node-id 2;
        address ipv4 172.16.2.43:7000;
        volume 0 {
            device minor 100;
            disk /dev/drbdpool/rdqmapp_00;
            meta-disk internal;
        }
    }
    connection-mesh {
        hosts rdqm-b1 rdqm-b2 rdqm-b3;
    }
}
